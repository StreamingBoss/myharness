cmake_minimum_required(VERSION 3.20)
project(myharness_tokenizer LANGUAGES C CXX)
include(FetchContent)
# The bridge never fetches dependencies. This explicit build step uses a fixed revision.
set(MYHARNESS_LLAMA_REVISION 3d65c90d04d337e88f2b1f7f0061f40a5324e662)
set(MYHARNESS_LLAMA_SOURCE "" CACHE PATH "Existing checkout of the pinned llama.cpp revision")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
if(MYHARNESS_LLAMA_SOURCE)
  execute_process(COMMAND git -C "${MYHARNESS_LLAMA_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE source_revision OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT source_revision STREQUAL MYHARNESS_LLAMA_REVISION)
    message(FATAL_ERROR "MYHARNESS_LLAMA_SOURCE must be the pinned revision ${MYHARNESS_LLAMA_REVISION}")
  endif()
  set(llama_cpp_SOURCE_DIR "${MYHARNESS_LLAMA_SOURCE}")
  add_subdirectory("${llama_cpp_SOURCE_DIR}" llama.cpp)
else()
  FetchContent_Declare(llama_cpp GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git GIT_TAG ${MYHARNESS_LLAMA_REVISION})
  FetchContent_MakeAvailable(llama_cpp)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(myharness-tokenizer main.cpp)
target_compile_features(myharness-tokenizer PRIVATE cxx_std_17)
target_include_directories(myharness-tokenizer PRIVATE "${llama_cpp_SOURCE_DIR}/vendor")
target_link_libraries(myharness-tokenizer PRIVATE llama)
install(TARGETS myharness-tokenizer RUNTIME DESTINATION bin)
